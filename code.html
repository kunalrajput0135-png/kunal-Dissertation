# 19. TEXT PREPROCESSING FUNCTION

def clean_text(text):

    text = str(text)

    # Lowercase
    text = text.lower()

    # Remove HTML
    text = re.sub(
        r"<[^>]+>",
        " ",
        text
    )

    # Remove URLs
    text = re.sub(
        r"http\S+|www\S+",
        " ",
        text
    )

    # Preserve letters and numbers
    text = re.sub(
        r"[^a-zA-Z0-9\s]",
        " ",
        text
    )

    # Remove repeated whitespace
    text = re.sub(
        r"\s+",
        " ",
        text
    ).strip()

    return text


model_df["Clean_Review"] = (
    model_df["Combined_Review"]
    .apply(clean_text)

# 43. FACTOR FREQUENCY ANALYSIS

factor_results = []


for factor, keywords in hotel_factors.items():

    positive_count = 0

    negative_count = 0

    positive_reviews = (
        model_df.loc[
            model_df["Sentiment"] == 1,
            "Clean_Review"
        ]
    )

    negative_reviews = (
        model_df.loc[
            model_df["Sentiment"] == 0,
            "Clean_Review"
        ]
    )


    for keyword in keywords:

        pattern = (
            r"\b"
            + re.escape(
                keyword.lower()
            )
            + r"\b"
        )

        positive_count += (
            positive_reviews
            .str.contains(
                pattern,
                regex=True,
                na=False
            )
            .sum()
        )

        negative_count += (
            negative_reviews
            .str.contains(
                pattern,
                regex=True,
                na=False
            )
            .sum()
        )


    factor_results.append({

        "Factor":
            factor,

        "Positive_Count":
            positive_count,

        "Negative_Count":
            negative_count
    })


factor_df = pd.DataFrame(
    factor_results
)
